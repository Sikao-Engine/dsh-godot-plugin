cmake_minimum_required(VERSION 3.24)
project(graphics101 VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Graphics101 requires Windows x64")
endif()
find_package(Vulkan REQUIRED COMPONENTS glslc)
include(FetchContent)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_MakeAvailable(nlohmann_json)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/build-toolchain.json" CONTENT
  "{\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"windowsSDK\":\"${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}\",\"vulkanSDK\":\"${Vulkan_VERSION}\",\"cmake\":\"${CMAKE_VERSION}\"}\n")
add_executable(graphics101 src/main.cpp src/camera.cpp src/vulkan_renderer.cpp src/interop_probe.cpp src/shared_surface_win32.cpp src/worker_protocol.cpp)
target_compile_features(graphics101 PRIVATE cxx_std_20)
set_property(TARGET graphics101 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(graphics101 PRIVATE VK_USE_PLATFORM_WIN32_KHR WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(graphics101 PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(graphics101 PRIVATE Vulkan::Vulkan nlohmann_json::nlohmann_json d3d11 dxgi user32)
foreach(stage vert frag)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/triangle.${stage}.spv")
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND Vulkan::glslc "${CMAKE_CURRENT_SOURCE_DIR}/shaders/triangle.${stage}" -o "${output}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/triangle.${stage}" VERBATIM)
  list(APPEND shader_outputs "${output}")
endforeach()
add_custom_target(graphics101_shaders ALL DEPENDS ${shader_outputs})
add_dependencies(graphics101 graphics101_shaders)
add_custom_command(TARGET graphics101 POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders" "$<TARGET_FILE_DIR:graphics101>/shaders")
enable_testing()
add_test(NAME camera COMMAND graphics101 --camera-test)
add_test(NAME interop_probe COMMAND graphics101 --probe)
add_test(NAME native_smoke COMMAND graphics101 --smoke --validation --frames 180)
set_tests_properties(native_smoke PROPERTIES TIMEOUT 120)
